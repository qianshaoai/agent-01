#### 新增

```ts
type Option = { value: string; label: string };

function buildProviderOptions(providers: Provider[]): Option[] {
  return providers
    .filter((p) => p.enabled && p.has_api_key)
    .map((p) => ({ value: p.id, label: `${p.name}（${p.provider_code}）` }));
}

function buildModelsForProvider(provider: Provider | undefined): Option[] {
  if (!provider) return [];
  const r = resolveProviderPresetForBuilder(provider);
  if (r.kind === "recognized" && r.preset.recommendedModels?.length) {
    return r.preset.recommendedModels.map((m) => ({ value: m.value, label: stripModelDesc(m.label) }));
  }
  if (r.kind === "custom") {
    const all = getPresetsByCategory("model");
    if (provider.platform === "openai") {
      return (all.find((p) => p.code === "openai-official")?.recommendedModels ?? [])
        .map((m) => ({ value: m.value, label: stripModelDesc(m.label) }));
    }
    if (provider.platform === "anthropic") {
      return (all.find((p) => p.code === "anthropic-official")?.recommendedModels ?? [])
        .map((m) => ({ value: m.value, label: stripModelDesc(m.label) }));
    }
  }
  return [];
}
```

## 人工待跑（dev 实测 5 项）

1. 新建 chat 智能体 → 一级看到供应商列表 → 二级 disabled + placeholder「请先选供应商」
2. 选某供应商 → 二级出现该 provider 的 recommendedModels → 选定 → draft.model_params.model 写入
3. 切到另一供应商 → 二级列表换 + model 字段 reset 为空 → 不静默错配
4. 老 draft（已有 provider + model）打开 → 一级 + 二级正确回显（model 在 recommendedModels 命中）
5. 没可用供应商 → 一级 disabled + amber 提示去模型接入
